raw scored data and diff against what the .docx actually prints. For all six
tools, not just SCAR. Prints one PASS/FAIL line per checked cell group.

Sources of truth:
  scored_results_render_hash.csv   -> Tables 3,5,7,9,10 (per-image aggregate)
  per_document_recovery.csv        -> Table 8 (depth), and an independent
                                      recomputation of Table 4 and Table 11
  c1_capability_gap.csv            -> Table 4 (primary)
  c2_repair_ablation.csv           -> Table 6
  mcnemar_results.csv              -> Table 11
"""

for r in scored:
    for k in ("GT_docs", "Faithful", "Partial", "Duplicate", "False_accept", "Junk"):
        r[k] = int(r[k])

_fail = [0]


def _norm(x):
    """Normalise a scalar or tuple so 90 == 90.0 for comparison."""
    if isinstance(x, (tuple, list)):
        return tuple(_norm(v) for v in x)
    try:
        return round(float(x), 4)
    except (TypeError, ValueError):
        return x


def check(label, got, want):
    ok = (_norm(got) == _norm(want))
    if not ok:
        _fail[0] += 1
    print(("  PASS " if ok else "  FAIL ") + "%-52s got=%s want=%s"
          % (label, got, want))


def recall(f, g):
    return round(f / g * 100, 1) if g else 0.0


def fa_rate(fa, f, p):
    d = f + p + fa
    return round(fa / d * 100, 1) if d else 0.0


def agg(images_pred, bucket):
    """bucket(image)->label; returns {(label,tool): dict of summed counts}."""
    out = defaultdict(lambda: defaultdict(int))
    for r in scored:
        if not images_pred(r["Image"]):
            continue
        b = bucket(r["Image"])
        if b is None:
            continue
        key = (b, r["Tool"])
        for k in ("GT_docs", "Faithful", "Partial", "Duplicate", "False_accept", "Junk"):
            out[key][k] += r[k]
    return out
